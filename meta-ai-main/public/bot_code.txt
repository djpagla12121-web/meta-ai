def parse_proxy_dict(proxy_str):
    if not proxy_str:
        return None
    s = proxy_str.strip()
    if not s.startswith("http://") and not s.startswith("https://") and not s.startswith("socks5://"):
        s = f"http://{s}"
    return {
        "http": s,
        "https": s
    }

def test_proxy(proxy_str):
    try:
        proxies = parse_proxy_dict(proxy_str)
        r = requests.get("https://api.ipify.org?format=json", proxies=proxies, timeout=10)
        if r.status_code == 200:
            return r.json().get("ip")
    except Exception as e:
        print(f"Proxy test error: {e}")
    return None

def login_to_yesms(force=False):
    global CURRENT_SESSION_COOKIE
    try:
        if not force and CURRENT_SESSION_COOKIE:
            return True
        login_data = {'identifier': YESMS_EMAIL, 'password': YESMS_PASSWORD}
        headers = {
            'User-Agent': 'Mozilla/5.0 (Linux; Android 12) AppleWebKit/537.36 Chrome/120.0.0.0 Mobile Safari/537.36',
            'Content-Type': 'application/x-www-form-urlencoded',
            'Origin': BASE_URL,
            'X-Requested-With': 'mark.via.gp'
        }
        res = session.post(f"{BASE_URL}/login", data=login_data, headers=headers, timeout=25, verify=False)
        if res.status_code == 200:
            for cookie in session.cookies:
                if cookie.name == 'session':
                    CURRENT_SESSION_COOKIE = cookie.value
                    return True
        return False
    except Exception as e:
        print(f"Login error: {e}")
        return False

def allocate_number(range_id):
    if not login_to_yesms():
        return None
    headers = {
        'User-Agent': 'Mozilla/5.0 (Linux; Android 12)',
        'Content-Type': 'application/json',
        'Origin': BASE_URL,
        'Cookie': f'session={CURRENT_SESSION_COOKIE}'
    }
    try:
        res = session.post(f"{BASE_URL}/api/allocate_number", json={'range_id': range_id}, headers=headers, timeout=25, verify=False)
        if res.status_code == 200:
            d = res.json()
            return d.get('number') or (d.get('data') and d['data'].get('no_plus_number')) or d
    except Exception as e:
        print(f"Allocate error: {e}")
    return None

@bot.callback_query_handler(func=lambda call: True)
def handle_callback(call):
    chat_id = call.message.chat.id
    msg_id = call.message.message_id
    data = call.data
    bot.answer_callback_query(call.id)

    if data.startswith("get_number_"):
        service_name = data.replace("get_number_", "")
        for s in SERVICES:
            if s.get("name") == service_name:
                ranges = s.get("ranges", [])
                markup = types.InlineKeyboardMarkup(row_width=2)
                for r in ranges:
                    country = s.get("country", {}).get(r, "Unknown")
                    markup.add(types.InlineKeyboardButton(f"📍 {country}", callback_data=f"allocate_{service_name}_{r}"))
                bot.edit_message_text(f"🔍 Select Country for {service_name}:", chat_id, msg_id, reply_markup=markup)
                return

    if data.startswith("allocate_"):
        parts = data.split("_")
        service_name = parts[1]
        range_id = "_".join(parts[2:])
        country = "Madagascar"
        for s in SERVICES:
            if s.get("name") == service_name:
                country = s.get("country", {}).get(range_id, "Madagascar")
                break

        bot.edit_message_text(f"⏳ Getting number from {country}...", chat_id, msg_id)
        result = allocate_number(range_id)
        if result:
            number = str(result)
            ACTIVE_BOOKINGS[number] = {
                "chat_id": chat_id,
                "number": number,
                "service": service_name,
                "range": range_id,
                "country": country,
                "time": datetime.now(),
                "otp": None
            }
            msg_text = f"<b>NUMBER</b> {number}\n<b>COUNTRY</b> {country.upper()}"
            bot.edit_message_text(msg_text, chat_id, msg_id, parse_mode="HTML")
        else:
            bot.edit_message_text("❌ Failed to get number! Please try again later.", chat_id, msg_id)

def otp_monitor_loop():
    while True:
        try:
            if login_to_yesms():
                numbers = get_user_numbers()
                if numbers:
                    for num in numbers:
                        no_plus = num.get('no_plus_number', '')
                        otp_code = num.get('otp_code', '')
                        if num.get('status') == 'success' and otp_code:
                            otp_key = f"{no_plus}_{otp_code}"
                            if otp_key not in SENT_OTPS and no_plus in ACTIVE_BOOKINGS:
                                item = ACTIVE_BOOKINGS[no_plus]
                                item["otp"] = otp_code
                                SENT_OTPS.add(otp_key)
                                msg_text = f"<b>NUMBER</b> {no_plus}\n<b>OTP CODE</b> {otp_code}"
                                try:
                                    bot.send_message(item["chat_id"], msg_text, parse_mode="HTML")
                                except Exception as e:
                                    print(f"Error sending OTP: {e}")
        except Exception as e:
            print(f"OTP Loop error: {e}")
        time.sleep(5)

if __name__ == "__main__":
    print("Starting bot...")
    threading.Thread(target=otp_monitor_loop, daemon=True).start()
    bot.infinity_polling(timeout=60)
